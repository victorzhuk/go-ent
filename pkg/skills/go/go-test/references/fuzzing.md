## Basic Fuzzing

```go
func FuzzParse(f *testing.F) {
    // Seed corpus - starting points for fuzzer
    f.Add("valid input")
    f.Add("")
    f.Add("extreme-case-12345")

    f.Fuzz(func(t *testing.T, input string) {
        result, err := Parse(input)
        if err != nil {
            return  // Expected for invalid input
        }
        // Validate invariants
        if result == "" {
            t.Error("empty result from valid parse")
        }
    })
}

// Run: go test -fuzz=FuzzParse -fuzztime=30s
```

## Multiple Seeds

```go
func FuzzJSONParse(f *testing.F) {
    // Provide diverse starting points
    f.Add(`{"id":1,"name":"test"}`)
    f.Add(`{}`)
    f.Add(`{"nested":{"field":"value"}}`)
    f.Add(`[]`)   // Invalid for this parser
    f.Add(`null`)

    f.Fuzz(func(t *testing.T, data string) {
        var result map[string]interface{}
        err := json.Unmarshal([]byte(data), &result)
        if err != nil {
            return  // Expected for malformed JSON
        }
        // Validate parsed data
        if result == nil {
            t.Error("null result after successful parse")
        }
    })
}
```

## Complex Types

```go
// Fuzz primitives, build complex types in function
func FuzzCreateUser(f *testing.F) {
    f.Add("alice@example.com", 25, "Alice")
    f.Add("", 0, "")

    f.Fuzz(func(t *testing.T, email string, age int, name string) {
        user := User{
            Email: email,
            Age:   age,
            Name:  name,
        }

        err := ValidateUser(user)
        if err != nil {
            return  // Invalid input is expected
        }

        // Verify invariants for valid users
        if user.Email == "" || user.Age < 0 {
            t.Error("validator accepted invalid user")
        }
    })
}
```

## Corpus Management

```bash
# Create corpus directory
mkdir -p testdata/fuzz/FuzzJSONParse

# Add manual seed
echo '{"special":"case"}' > testdata/fuzz/FuzzJSONParse/seed_custom

# Directory structure:
# testdata/fuzz/FuzzJSONParse/
# ├── 1a2b3c4d5e6f7890  # Auto-generated by fuzzer
# ├── a1b2c3d4e5f6g7h8  # Another finding
# └── seed_custom      # Manual seed
```

## Running Fuzzing

```bash
# Run locally (30 seconds)
go test -fuzz=FuzzParse -fuzztime=30s

# Run all fuzz tests in CI (10 seconds each)
go test -fuzz=. -fuzztime=10s

# With coverage
go test -fuzz=FuzzParse -fuzztime=1m -coverprofile=fuzz.out

# Only run fuzzing on specific package
go test -fuzz=FuzzParse ./internal/parser
```
